   function setLanguage(lang) {

     document.getElementById('btn-ja').classList.toggle('active', lang === 'ja');

     document.getElementById('btn-zh').classList.toggle('active', lang === 'zh');

   

     document.querySelectorAll('[data-key]').forEach(el => {

       const key = el.getAttribute('data-key');

       if (i18n[lang][key]) {

         el.innerHTML = i18n[lang][key];

       }

     });

   }

   function handleFormSubmit(event) {

     event.preventDefault();

     // ブラウザのポップアップブロック対策として、クリックイベント直後に空の別タブを開いておく

     var newTab = window.open('about:blank', '_blank');

     var btn = document.getElementById('submitBtn');

     btn.disabled = true;

     btn.innerText = '送信中 / 正在提交...';

     var formData = {

       name: event.target.name.value,

       email: event.target.email.value,

       phone: event.target.phone.value,

       agreed: event.target.agreed.checked

     };

     google.script.run

       .withSuccessHandler(function(response) {

         if (response.status === 'success') {

           // 事前に開いておいた別タブを eeasy の日程調整URLに遷移させる

           newTab.location.href = response.redirectUrl;

         

           btn.innerText = '日程調整ページを開きました（別タブ）';

         } else {

           // エラー時は開いた空タブを閉じる

           if (newTab) newTab.close();

           alert('送信に失敗しました: ' + response.message);

           btn.disabled = false;

           btn.innerText = 'プロに無料相談する（日程調整へ）';

         }

       })

       .withFailureHandler(function(error) {

         if (newTab) newTab.close();

         alert('通信エラーが発生しました: ' + error.message);

         btn.disabled = false;

         btn.innerText = 'プロに無料相談する（日程調整へ）';

       })

       .processForm(formData);

   }
